    val inOutCount = editingContext.getInOuts().size
    if (inOutCount < MIN_INOUT_ELEMENTS) {  // MIN_INOUT_ELEMENTS = 2

                "the railway network.\n" +
                "Please add more InOut elements before saving.",
            "Cannot Save - Insufficient InOut Elements",  // Dialog title

        )
